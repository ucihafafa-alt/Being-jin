rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isAdmin() {
      return request.auth != null && request.auth.uid in [
        'ADMIN_UID_ЭНД_БИЧНЭ'
      ];
    }

    match /requests/{requestId} {
      // Хэрэглэгч хүсэлт үүсгэж болно.
      allow create: if true;

      // Админ бүх хүсэлтийг харна. Хэрэглэгч зөвхөн баталгаажсан тайлангийн link-ээр уншина.
      allow get: if isAdmin() || resource.data.status == 'approved';
      allow list: if isAdmin();

      // Зөвхөн админ батлах/засах боломжтой.
      allow update, delete: if isAdmin();
    }
  }
}
